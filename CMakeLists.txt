cmake_minimum_required(VERSION 3.24)
project(SupleqNativeBuffer LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NATIVE_TESTS "Build portable scheduler tests" ON)
if(NATIVE_TESTS)
  enable_testing()
  add_executable(buffer_scheduler_tests tests/test_timing.cpp)
  add_test(NAME buffer_scheduler_tests COMMAND buffer_scheduler_tests)
endif()
if(WIN32)
  include(FetchContent)
  FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(minhook)
  add_library(SupleqBufferHook SHARED src/helper.cpp)
  target_link_libraries(SupleqBufferHook PRIVATE minhook)
  target_compile_definitions(SupleqBufferHook PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(SupleqBufferHook PROPERTIES OUTPUT_NAME "SupleqBufferHook")
  add_executable(SupleqInputBuffer src/launcher.cpp)
  target_compile_definitions(SupleqInputBuffer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(SupleqInputBuffer PRIVATE version)
endif()
